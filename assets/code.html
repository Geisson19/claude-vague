1// Runs fn once the calls stop for ms milliseconds
2export function debounce<T extends unknown[]>(fn: (...args: T) => void, ms = 200) {
3  let timer: ReturnType<typeof setTimeout> | undefined
4
5  return (...args: T) => {
6    clearTimeout(timer)
7    timer = setTimeout(() => fn(...args), ms)
8  }
9}
